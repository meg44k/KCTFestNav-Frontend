# クラス展示のいいね(人気投票) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 来場者がクラス展示に端末ごと 1 回いいねでき、管理者と学生会が数・推移を見て、急に増えた時間帯のいいねを取り消せるようにする。

**Architecture:** バックエンドに `likes`・`like_removals` 表、署名付きの投票者番号(`internal/auth/voter.go`)、Redis の回数上限、`/internal/*`(合言葉 `X-Internal-Key` で Next.js のサーバーだけが呼ぶ)と `/manage/likes`(JWT)を足す。フロントは Server Action が cookie `kct-voter` の番号と合言葉を付けて呼び、`LikesProvider` が自分のいいねを持って `BoothCard` の ♡ を描く。管理画面に「いいね」(順位・10 分ごとの推移・⚠・時間帯の取り消し)を足す。

**Tech Stack:** Go 1.25 / Echo v5 / sqlc / MySQL 8.4 / go-redis v9 / testify、Next.js 16 / React 19 / Server Actions / vitest、Terraform

**Spec:** `docs/superpowers/specs/2026-10-09-likes-design.md`(KCTFestNav-Frontend)

## Global Constraints

- 対象はクラス展示だけ: 主催者が `^(\d+)\s*-\s*\d+$` で学年 1〜5(フロントの `parseGrade` と同じ)
- 1 つの投票者番号で同じブースは 1 件。取り消せる
- 来場者には数を見せない。数・推移は Admin と Gakuseikai だけ。全部消すは Admin だけ
- いつでも押せる。IP は使わない
- 投票者番号の形 `<UUID>.<署名>`、署名は HMAC-SHA256(`VOTER_SECRET`)を base64url(パディング無し)
- cookie `kct-voter`: `httpOnly`・`secure`・`sameSite=lax`・`path=/`・90 日
- 上限: 番号を作るのはサーバー全体で 1 分 300 個、切り替えは番号ごとに 1 分 60 回。超えたら 429、文言「混み合っています。少し待ってからもう一度押してください」
- ⚠: 10 分ごとの数が 30 件以上かつ、そのブースの 10 分ごとの数(0 を除く)の中央値の 5 倍以上
- 時刻は UTC で保存し、画面で JST
- 本人のサーバー(:1323、:3000)と本人の開発 DB `kctfestnav` は触らない。テスト DB(`kctfest_test_handler`・`kctfest_test_repository`)にはマイグレーションを流してよい
- ブランチ: 両リポジトリとも develop から `feature/likes`

## Review Focus

1. **番号の偽造と使い回し**: 書き換えた署名・別の秘密で作った番号・`.` の無い文字列・空文字は全部 401(Task 1・5)
2. **合言葉の無い直接の呼び出し**: `/internal/*` に `X-Internal-Key` が無い・違うと 401。`INTERNAL_API_KEY` が空の設定でも「空と空で一致」して通らないこと(Task 5)
3. **クラス展示でないブースへの PUT**(バザー・存在しない ID)は 400 / 404(Task 5)
4. **同時に押されたとき**: 同じ番号で PUT を 2 回続けても 1 件(主キーで弾き `INSERT IGNORE`)(Task 3・5)
5. **時間帯の取り消しの境界**: `from` 以上 `to` 未満。`from >= to` は 400、JST で選んだ時間が UTC で正しく送られる(Task 5・9)

```go
package auth

import (
	"crypto/hmac"
	"crypto/sha256"
	"encoding/base64"
	"errors"
	"strings"

	"github.com/google/uuid"
)

// 来場者のいいねに使う番号。<UUID>.<署名>。番号を自分で作れないようにサーバーが署名する
var ErrInvalidVoter = errors.New("invalid voter")

func sign(id string, secret []byte) string {
	m := hmac.New(sha256.New, secret)
	m.Write([]byte(id))
	return base64.RawURLEncoding.EncodeToString(m.Sum(nil))
}

func NewVoterToken(secret []byte) (string, string) {
	id := uuid.NewString()
	return id + "." + sign(id, secret), id
}

func ParseVoterToken(token string, secret []byte) (string, error) {
	if len(secret) == 0 {
		return "", ErrInvalidVoter
	}
	id, sig, ok := strings.Cut(token, ".")
	if !ok {
		return "", ErrInvalidVoter
	}
	if _, err := uuid.Parse(id); err != nil {
		return "", ErrInvalidVoter
	}
	if !hmac.Equal([]byte(sig), []byte(sign(id, secret))) {
		return "", ErrInvalidVoter
	}
	return id, nil
}
```

package domain

import (
	"context"
	"regexp"
	"sort"
	"strconv"
	"time"
)

var classOrganizer = regexp.MustCompile(`^(\d+)\s*-\s*\d+$`)

// クラス展示の主催者("1-1" のようなクラス表記、学年 1〜5)か。フロントの parseGrade と同じ決まり
func IsClassOrganizer(organizer string) bool {
	m := classOrganizer.FindStringSubmatch(trimSpace(organizer))
	if m == nil {
		return false
	}
	g, _ := strconv.Atoi(m[1])
	return g >= 1 && g <= 5
}

func trimSpace(s string) string {
	for len(s) > 0 && s[0] == ' ' {
		s = s[1:]
	}
	for len(s) > 0 && s[len(s)-1] == ' ' {
		s = s[:len(s)-1]
	}
	return s
}

// 10 分ごとのいいねの数
type LikeBucket struct {
	Start time.Time
	Count int
	Burst bool
}

type BoothLikes struct {
	BoothID   int
	Name      string
	Organizer string
	Total     int
	Buckets   []LikeBucket
	Burst     bool
}

const (
	burstMin   = 30
	burstRatio = 5
)

// 10 分の数が 30 件以上で、0 を除いた中央値の 5 倍以上なら ⚠
func MarkBursts(buckets []LikeBucket) []LikeBucket {
	out := make([]LikeBucket, len(buckets))
	copy(out, buckets)
	var nonzero []int
	for _, x := range out {
		if x.Count > 0 {
			nonzero = append(nonzero, x.Count)
		}
	}
	if len(nonzero) == 0 {
		return out
	}
	sort.Ints(nonzero)
	median := float64(nonzero[len(nonzero)/2])
	if len(nonzero)%2 == 0 {
		median = float64(nonzero[len(nonzero)/2-1]+nonzero[len(nonzero)/2]) / 2
	}
	for i := range out {
		out[i].Burst = out[i].Count >= burstMin && float64(out[i].Count) >= median*burstRatio
	}
	return out
}

type LikeRepository interface {
	Like(ctx context.Context, boothID int, voterID string, at time.Time) error
	Unlike(ctx context.Context, boothID int, voterID string) error
	Mine(ctx context.Context, voterID string) ([]int, error)
	// 全ブースのいいねの時刻(集計は usecase で 10 分ごとにまとめる)
	AllLikeTimes(ctx context.Context) (map[int][]time.Time, error)
	RemoveRange(ctx context.Context, boothID int, from, to time.Time, by string, at time.Time) (int, error)
	RemoveAll(ctx context.Context, by string, at time.Time) (int, error)
	// 上限。超えていなければ true
	AllowNewVoter(ctx context.Context) (bool, error)
	AllowToggle(ctx context.Context, voterID string) (bool, error)
}
```

func (r *likeRepository) allow(ctx context.Context, key string, limit int64) (bool, error) {
	n, err := r.rdb.Incr(ctx, key).Result()
	if err != nil {
		return false, err
	}
	if n == 1 {
		r.rdb.Expire(ctx, key, 2*time.Minute)
	}
	return n <= limit, nil
}

func minuteKey(t time.Time) string { return t.UTC().Format("200601021504") }

func (r *likeRepository) AllowNewVoter(ctx context.Context) (bool, error) {
	return r.allow(ctx, "likes:new-voters:"+minuteKey(r.now()), 300)
}

func (r *likeRepository) AllowToggle(ctx context.Context, voterID string) (bool, error) {
	return r.allow(ctx, "likes:toggle:"+voterID+":"+minuteKey(r.now()), 60)
}
```

```go
func InternalKey(key string) echo.MiddlewareFunc {
	return func(next echo.HandlerFunc) echo.HandlerFunc {
		return func(c *echo.Context) error {
			got := c.Request().Header.Get("X-Internal-Key")
			if key == "" || subtle.ConstantTimeCompare([]byte(got), []byte(key)) != 1 {
				return echo.NewHTTPError(http.StatusUnauthorized, "unauthorized")
			}
			return next(c)
		}
	}
}
```
